### 7.3 安全隔离实现

```python
# src/services/opencode/security.py

from pathlib import Path
from typing import List, Set


class OpenCodeSecurityGuard:
    """OpenCode 安全隔离器"""

    BLOCKED_EXTENSIONS: Set[str] = {
        ".exe", ".bat", ".cmd", ".ps1", ".sh",
        ".dll", ".so", ".dylib"
    }

    BLOCKED_COMMANDS: Set[str] = {
        "rm -rf", "format", "del /s", "rd /s",
        "sudo", "chmod 777", "wget", "curl"
    }

    def __init__(
        self,
        allowed_dirs: List[str],
        max_file_size: int = 100 * 1024 * 1024
    ):
        self.allowed_dirs = [Path(d).resolve() for d in allowed_dirs]
        self.max_file_size = max_file_size

    def validate_path(self, path: str) -> bool:
        """验证路径是否在允许范围内"""
        resolved = Path(path).resolve()

        for allowed in self.allowed_dirs:
            if str(resolved).startswith(str(allowed)):
                return True

        return False

    def validate_extension(self, path: str) -> bool:
        """验证文件扩展名"""
        ext = Path(path).suffix.lower()
        return ext not in self.BLOCKED_EXTENSIONS

    def validate_command(self, command: str) -> bool:
        """验证命令是否安全"""
        cmd_lower = command.lower().strip()

        for blocked in self.BLOCKED_COMMANDS:
            if blocked in cmd_lower:
                return False

        return True

    def validate_size(self, size: int) -> bool:
        """验证文件大小"""
        return size <= self.max_file_size
```

from typing import Optional, Any


class OpenCodeError(Exception):
    """OpenCode 基础异常"""

    def __init__(self, message: str, details: Optional[dict] = None):
        super().__init__(message)
        self.message = message
        self.details = details or {}


class OpenCodeConnectionError(OpenCodeError):

    """会话错误"""

class OpenCodeTimeoutError(OpenCodeError):

### 8.2 重试装饰器

```python
import asyncio
import httpx
from functools import wraps
from typing import TypeVar, Callable


T = TypeVar('T')


def with_retry(
    max_retries: int = 3,
    retry_delay: float = 1.0,
    backoff_factor: float = 2.0,
    retryable_exceptions: tuple = (httpx.HTTPError, asyncio.TimeoutError)
):
    """
    重试装饰器

    - max_retries: 最大重试次数
    - retry_delay: 初始延迟（秒）
    - backoff_factor: 退避因子
    """
    def decorator(func: Callable[..., T]) -> Callable[..., T]:
        @wraps(func)
        async def wrapper(*args, **kwargs) -> T:
            last_exception = None
            current_delay = retry_delay

            for attempt in range(max_retries + 1):
                try:
                    return await func(*args, **kwargs)
                except retryable_exceptions as e:
                    last_exception = e
                    if attempt < max_retries:
                        print(f"[重试] {func.__name__} 失败 ({attempt+1}/{max_retries}): {e}")
                        await asyncio.sleep(current_delay)
                        current_delay *= backoff_factor
                    else:
                        print(f"[重试] {func.__name__} 达到最大重试次数")

            raise last_exception
        return wrapper
    return decorator
```

### 8.3 错误恢复策略

| 错误类型           | 恢复策略           | 说明                 |
| ------------------ | ------------------ | -------------------- |
| connection_refused | restart_server     | 重启 OpenCode 服务器 |
| timeout            | retry_with_backoff | 带退避的重试         |
| session_error      | recreate_session   | 重新创建会话         |
| permission_denied  | check_permission   | 检查权限配置         |
| resource_exhausted | cleanup_and_retry  | 清理资源后重试       |
