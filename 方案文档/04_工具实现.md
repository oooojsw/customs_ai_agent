## 5. 工具实现

### 5.1 工具注册架构

参考现有 MCP Bridge 的架构，将 OpenCode 的功能封装为 LangChain StructuredTool。

### 5.2 核心工具列表

| 工具名称                | 功能描述           | 对应 API                              |
| ----------------------- | ------------------ | ------------------------------------- |
| opencode_create_session | 创建 OpenCode 会话 | POST /session                         |
| opencode_send_message   | 发送消息到会话     | POST /session/:id/message             |
| opencode_read_file      | 读取文件内容       | GET /file/content                     |
| opencode_write_file     | 写入文件内容       | (通过 session 工具)                   |
| opencode_search_text    | 搜索文本           | GET /find                             |
| opencode_search_files   | 搜索文件名         | GET /find/file                        |
| opencode_list_files     | 列出目录内容       | GET /file                             |
| opencode_run_command    | 执行命令           | POST /session/:id/message (bash tool) |
| opencode_get_messages   | 获取会话消息       | GET /session/:id/message              |
| opencode_delete_session | 删除会话           | DELETE /session/:id                   |

### 5.3 工具实现代码

#### 5.3.1 opencode_tools.py

```python
"""
OpenCode 工具注册模块
将 OpenCode HTTP API 封装为 LangChain StructuredTool
"""
from langchain_core.tools import Tool, StructuredTool
from pydantic import Field, create_model
from typing import Optional, List, Dict, Any
import json


class OpenCodeTools:
    """OpenCode 工具管理器"""

    def __init__(self, client):
        """
        初始化工具管理器
        :param client: OpenCodeClient 实例
        """
        self.client = client
        self.tools: List[Any] = []
        self._register_tools()

    def _register_tools(self):
        """注册所有工具"""

        # ========== 1. 会话管理工具 ==========

        async def create_session_tool(input_str: str) -> str:
            """
            创建 OpenCode 会话

            参数格式: "workspace_dir|title"
            示例: "data|我的会话"
            """
            try:
                parts = input_str.split("|", 1)
                workspace_dir = parts[0].strip()
                title = parts[1].strip() if len(parts) > 1 else None

                result = await self.client.create_session(
                    workspace_dir=workspace_dir,
                    title=title
                )

                return json.dumps({
                    "success": True,
                    "session_id": result["id"],
                    "title": result["title"],
                    "directory": result["directory"]
                }, ensure_ascii=False)
            except Exception as e:
                return json.dumps({
                    "success": False,
                    "error": str(e)
                }, ensure_ascii=False)

        self.tools.append(Tool(
            name="opencode_create_session",
            func=lambda x: "此工具仅支持异步环境运行",
            coroutine=create_session_tool,
            description="""创建 OpenCode 会话用于文件操作和代码执行。

使用时机：需要进行文件操作、代码执行、程序编辑时。

参数格式: "workspace_dir|title"（title 可选）
示例: "data|文件处理会话"

返回: 会话ID 和相关信息
"""
        ))

        # ========== 2. 消息发送工具 ==========

        async def send_message_tool(input_str: str) -> str:
            """
            发送消息到 OpenCode 会话

            参数格式: "session_id|message"
            示例: "sess_xxxx|读取 config.json 文件"
            """
            try:
                parts = input_str.split("|", 1)
                if len(parts) != 2:
                    return "错误：参数格式应为 'session_id|message'"

                session_id = parts[0].strip()
                message = parts[1].strip()

                result = await self.client.send_message(
                    session_id=session_id,
                    parts=[{"type": "text", "text": message}]
                )

                # 提取助手回复
                parts_list = result.get("parts", [])
                response_text = ""
                for part in parts_list:
                    if part.get("type") == "text":
                        response_text += part.get("text", "")

                return response_text or "消息已发送，但未收到回复"
            except Exception as e:
                return f"发送消息失败: {str(e)}"

        self.tools.append(Tool(
            name="opencode_send_message",
            func=lambda x: "此工具仅支持异步环境运行",
            coroutine=send_message_tool,
            description="""发送消息到 OpenCode 会话。

使用时机：需要执行复杂操作、文件编辑、代码修改时。

参数格式: "session_id|message"
示例: "sess_xxxx|读取 config.json 文件内容"

注意：会话ID 由 opencode_create_session 返回
"""
        ))

        # ========== 3. 文件读取工具 ==========

        async def read_file_tool(path: str) -> str:
            """
            读取文件内容

            参数: 文件路径
            """
            try:
                result = await self.client.read_file(path.strip())
                content = result.get("content", "")
                return content if content else "文件内容为空"
            except Exception as e:
                return f"读取文件失败: {str(e)}"

        self.tools.append(Tool(
            name="opencode_read_file",
            func=lambda x: "此工具仅支持异步环境运行",
            coroutine=read_file_tool,
            description="""读取文件内容。

使用时机：需要查看文件内容时。

参数: 文件路径（相对于工作目录）
示例: "config/risk_rules.json"

注意：路径相对于 OpenCode 的工作目录
"""
        ))

        # ========== 4. 文件搜索工具 ==========

        async def search_files_tool(query: str) -> str:
            """
            搜索文件名

            参数: 搜索关键词
            """
            try:
                results = await self.client.search_files(query.strip())
                if not results:
                    return "未找到匹配的文件"
                return "\n".join(results)
            except Exception as e:
                return f"搜索文件失败: {str(e)}"

        self.tools.append(Tool(
            name="opencode_search_files",
            func=lambda x: "此工具仅支持异步环境运行",
            coroutine=search_files_tool,
            description="""按文件名搜索文件。

使用时机：需要查找特定文件时。

参数: 搜索关键词
示例: "config.json" 或 "risk_rules"

返回: 匹配的文件路径列表
"""
        ))

        # ========== 5. 文本搜索工具 ==========

        async def search_text_tool(pattern: str) -> str:
            """
            搜索文本内容（ripgrep）

            参数: 搜索正则表达式
            """
            try:
                results = await self.client.search_text(pattern.strip())
                if not results:
                    return "未找到匹配的文本"

                output = []
                for match in results[:10]:  # 限制返回数量
                    data = match.get("data", {})
                    file_path = data.get("path", {}).get("text", "")
                    line_text = data.get("lines", {}).get("text", "")
                    line_num = data.get("line_number", "")
                    output.append(f"{file_path}:{line_num}: {line_text.strip()}")

                return "\n".join(output)
            except Exception as e:
                return f"搜索文本失败: {str(e)}"

        self.tools.append(Tool(
            name="opencode_search_text",
            func=lambda x: "此工具仅支持异步环境运行",
            coroutine=search_text_tool,
            description="""在文件中搜索文本（使用正则表达式）。

使用时机：需要在代码或文档中查找特定内容时。

参数: 正则表达式模式
示例: "import.*json" 或 "def main"

返回: 匹配的文件、行号和内容
"""
        ))

        # ========== 6. 目录列表工具 ==========

        async def list_files_tool(path: str) -> str:
            """
            列出目录内容

            参数: 目录路径
            """
            try:
                results = await self.client.list_files(path.strip())
                if not results:
                    return "目录为空或不存在"

                output = []
                for item in results:
                    item_type = item.get("type", "unknown")
                    name = item.get("name", "")
                    size = item.get("size", "")

                    if item_type == "directory":
                        output.append(f"📁 {name}/")
                    else:
                        size_str = f" ({size} bytes)" if size else ""
                        output.append(f"📄 {name}{size_str}")

                return "\n".join(output)
            except Exception as e:
                return f"列出目录失败: {str(e)}"

        self.tools.append(Tool(
            name="opencode_list_files",
            func=lambda x: "此工具仅支持异步环境运行",
            coroutine=list_files_tool,
            description="""列出目录内容。

使用时机：需要查看目录结构时。

参数: 目录路径（"." 表示当前目录）
示例: "." 或 "config" 或 "src/services"

返回: 文件和子目录列表
"""
        ))

        # ========== 7. 命令执行工具 ==========

        async def run_command_tool(input_str: str) -> str:
            """
            执行 shell 命令

            参数格式: "session_id|command"
            示例: "sess_xxxx|ls -la"
            """
            try:
                parts = input_str.split("|", 1)
                if len(parts) != 2:
                    return "错误：参数格式应为 'session_id|command'"

                session_id = parts[0].strip()
                command = parts[1].strip()

                # 使用 shell 接口
                result = await self.client._client.post(
                    f"/session/{session_id}/shell",
                    json={
                        "agent": "general",
                        "command": command
                    }
                )
                result.raise_for_status()

                response_data = result.json()
                parts_list = response_data.get("parts", [])
                output = ""
                for part in parts_list:
                    if part.get("type") == "tool":
                        state = part.get("state", {})
                        if state.get("status") == "completed":
                            output += state.get("output", "")

                return output or "命令执行完成，无输出"
            except Exception as e:
                return f"执行命令失败: {str(e)}"

        self.tools.append(Tool(
            name="opencode_run_command",
            func=lambda x: "此工具仅支持异步环境运行",
            coroutine=run_command_tool,
            description="""执行 shell 命令。

使用时机：需要运行系统命令、脚本时。

参数格式: "session_id|command"
示例: "sess_xxxx|python script.py"

注意：需要先创建会话获取 session_id
"""
        ))

        # ========== 8. 获取消息工具 ==========

        async def get_messages_tool(session_id: str) -> str:
            """
            获取会话消息历史

            参数: 会话ID
            """
            try:
                messages = await self.client.get_messages(session_id.strip())
                if not messages:
                    return "会话中没有消息"

                output = []
                for msg in messages[-10:]:  # 只返回最近10条
                    role = msg.get("info", {}).get("role", "unknown")
                    parts = msg.get("parts", [])

                    text_parts = []
                    for part in parts:
                        if part.get("type") == "text":
                            text_parts.append(part.get("text", ""))

                    if text_parts:
                        output.append(f"[{role}] {''.join(text_parts)[:200]}")

                return "\n\n".join(output) if output else "没有可显示的消息"
            except Exception as e:
                return f"获取消息失败: {str(e)}"

        self.tools.append(Tool(
            name="opencode_get_messages",
            func=lambda x: "此工具仅支持异步环境运行",
            coroutine=get_messages_tool,
            description="""获取会话的消息历史。

使用时机：需要查看会话历史记录时。

参数: 会话ID

返回: 最近的消息列表
"""
        ))

        # ========== 9. 删除会话工具 ==========

        async def delete_session_tool(session_id: str) -> str:
            """
            删除会话

            参数: 会话ID
            """
            try:
                await self.client.delete_session(session_id.strip())
                return f"会话 {session_id} 已删除"
            except Exception as e:
                return f"删除会话失败: {str(e)}"

        self.tools.append(Tool(
            name="opencode_delete_session",
            func=lambda x: "此工具仅支持异步环境运行",
            coroutine=delete_session_tool,
            description="""删除 OpenCode 会话。

使用时机：操作完成后清理会话资源。

参数: 会话ID

返回: 删除结果
"""
        ))

    def get_tools(self) -> List[Any]:
        """获取所有注册的工具"""
        return self.tools
```
