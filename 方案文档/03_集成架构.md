### 4.3 核心组件设计

#### 4.3.1 OpenCode 配置 (opencode_config.py)

```python
from pydantic import BaseModel, Field
from typing import Optional
import json
from pathlib import Path


class OpenCodeConfig(BaseModel):
    """OpenCode 服务器配置"""
    enabled: bool = Field(True, description="是否启用 OpenCode 集成")
    port: int = Field(4096, description="OpenCode 服务器端口")
    hostname: str = Field("127.0.0.1", description="OpenCode 服务器主机名")
    server_password: Optional[str] = Field(None, description="HTTP Basic Auth 密码")
    server_username: str = Field("opencode", description="HTTP Basic Auth 用户名")
    timeout: int = Field(30, description="API 请求超时时间（秒）")
    workspace_dir: str = Field("data", description="默认工作目录")

    # OpenCode 源码目录（用 bun run dev serve 启动，不需要二进制文件）
    opencode_source_dir: str = Field(
        "",  # 必须配置，例如: C:\Users\xxx\Desktop\opencode re\opencode - copy
        description="OpenCode 源码项目路径"
    )

    # 代理设置
    http_proxy: Optional[str] = Field(None, description="HTTP 代理地址")


class OpenCodeConfigLoader:
    """OpenCode 配置加载器（单例）"""
    _instance = None
    _config: Optional[OpenCodeConfig] = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    def __init__(self):
        if self._config is None:
            self._load_config()

    def _load_config(self) -> None:
        """从 JSON 文件加载配置"""
        config_path = Path("data/opencode_config.json")

        if not config_path.exists():
            print("[OpenCodeConfig] 配置文件不存在，使用默认配置")
            self._config = OpenCodeConfig()
            return

        try:
            with open(config_path, "r", encoding="utf-8") as f:
                raw_config = json.load(f)
            self._config = OpenCodeConfig(**raw_config)
            print("[OpenCodeConfig] 配置加载成功")
        except Exception as e:
            print(f"[OpenCodeConfig] 配置加载失败: {e}")
            self._config = OpenCodeConfig()

    def get_config(self) -> OpenCodeConfig:
        """获取配置"""
        if self._config is None:
            self._load_config()
        return self._config

    def reload(self) -> None:
        """重新加载配置"""
        self._config = None
        self._load_config()


opencode_config_loader = OpenCodeConfigLoader()
```

#### 4.3.2 OpenCode HTTP 客户端 (opencode_client.py)

```python
import httpx
import asyncio
import json
from typing import Optional, Dict, Any, List
from pathlib import Path


class OpenCodeClient:
    """OpenCode 服务器 HTTP 客户端"""

    def __init__(
        self,
        base_url: str = "http://127.0.0.1:4096",
        username: str = "opencode",
        password: Optional[str] = None,
        timeout: int = 30,
        proxy: Optional[str] = None
    ):
        self.base_url = base_url.rstrip("/")
        self.timeout = timeout

        # 构建认证
        auth = None
        if password:
            auth = httpx.BasicAuth(username, password)

        # 构建传输层
        transport = None
        if proxy:
            transport = httpx.AsyncHTTPTransport(proxy=proxy)

        self._client = httpx.AsyncClient(
            base_url=self.base_url,
            auth=auth,
            transport=transport,
            timeout=httpx.Timeout(timeout)
        )

    # ==================== 会话管理 ====================

    async def create_session(
        self,
        workspace_dir: str,
        title: Optional[str] = None,
        parent_id: Optional[str] = None
    ) -> Dict[str, Any]:
        """创建新会话

        注意：工作目录通过 x-opencode-directory header 传递，不通过 body
        """
        body = {}
        if title:
            body["title"] = title
        if parent_id:
            body["parentID"] = parent_id

        response = await self._client.post(
            "/session",
            json=body,
            headers={"x-opencode-directory": workspace_dir}
        )
        response.raise_for_status()
        return response.json()

    async def list_sessions(
        self,
        directory: Optional[str] = None,
        roots: bool = False,
        limit: int = 50
    ) -> List[Dict[str, Any]]:
        """列出会话"""
        params = {"limit": limit}
        if directory:
            params["directory"] = directory
        if roots:
            params["roots"] = "true"

        response = await self._client.get("/session", params=params)
        response.raise_for_status()
        return response.json()

    async def get_session(self, session_id: str) -> Dict[str, Any]:
        """获取会话信息"""
        response = await self._client.get(f"/session/{session_id}")
        response.raise_for_status()
        return response.json()

    async def delete_session(self, session_id: str) -> bool:
        """删除会话"""
        response = await self._client.delete(f"/session/{session_id}")
        response.raise_for_status()
        return response.json()

    async def abort_session(self, session_id: str) -> bool:
        """中止会话"""
        response = await self._client.post(f"/session/{session_id}/abort")
        response.raise_for_status()
        return response.json()

    # ==================== 消息管理 ====================

    async def send_message(
        self,
        session_id: str,
        parts: List[Dict[str, Any]],
        model: Optional[Dict[str, str]] = None,
        agent: Optional[str] = None,
        system: Optional[str] = None
    ) -> Dict[str, Any]:
        """发送消息（非流式）"""
        body: Dict[str, Any] = {"parts": parts}
        if model:
            body["model"] = model
        if agent:
            body["agent"] = agent
        if system:
            body["system"] = system

        response = await self._client.post(
            f"/session/{session_id}/message",
            json=body
        )
        response.raise_for_status()
        return response.json()

    async def send_message_stream(
        self,
        session_id: str,
        parts: List[Dict[str, Any]],
        model: Optional[Dict[str, str]] = None,
        agent: Optional[str] = None
    ):
        """发送消息（SSE 流式）"""
        body: Dict[str, Any] = {"parts": parts}
        if model:
            body["model"] = model
        if agent:
            body["agent"] = agent

        async with self._client.stream(
            "POST",
            f"/session/{session_id}/message",
            json=body
        ) as response:
            response.raise_for_status()
            async for line in response.aiter_lines():
                if line.startswith("data: "):
                    try:
                        data = json.loads(line[6:])
                        yield data
                    except json.JSONDecodeError:
                        continue

    async def get_messages(
        self,
        session_id: str,
        limit: int = 50
    ) -> List[Dict[str, Any]]:
        """获取会话消息"""
        response = await self._client.get(
            f"/session/{session_id}/message",
            params={"limit": limit}
        )
        response.raise_for_status()
        return response.json()

    # ==================== 文件操作 ====================

    async def search_text(self, pattern: str) -> List[Dict[str, Any]]:
        """搜索文本（ripgrep）"""
        response = await self._client.get("/find", params={"pattern": pattern})
        response.raise_for_status()
        return response.json()

    async def search_files(
        self,
        query: str,
        file_type: Optional[str] = None,
        limit: int = 10
    ) -> List[str]:
        """搜索文件名"""
        params = {"query": query, "limit": limit}
        if file_type:
            params["type"] = file_type

        response = await self._client.get("/find/file", params=params)
        response.raise_for_status()
        return response.json()

    async def list_files(self, path: str) -> List[Dict[str, Any]]:
        """列出目录内容"""
        response = await self._client.get("/file", params={"path": path})
        response.raise_for_status()
        return response.json()

    async def read_file(self, path: str) -> Dict[str, Any]:
        """读取文件内容"""
        response = await self._client.get("/file/content", params={"path": path})
        response.raise_for_status()
        return response.json()

    async def get_file_status(self) -> List[Dict[str, Any]]:
        """获取文件 git 状态"""
        response = await self._client.get("/file/status")
        response.raise_for_status()
        return response.json()

    # ==================== 工具系统 ====================

    async def list_tools(
        self,
        provider: str,
        model: str
    ) -> List[Dict[str, Any]]:
        """列出可用工具"""
        response = await self._client.get(
            "/experimental/tool",
            params={"provider": provider, "model": model}
        )
        response.raise_for_status()
        return response.json()

    async def list_tool_ids(self) -> List[str]:
        """列出工具 ID"""
        response = await self._client.get("/experimental/tool/ids")
        response.raise_for_status()
        return response.json()

    # ==================== 配置管理 ====================

    async def get_config(self) -> Dict[str, Any]:
        """获取配置"""
        response = await self._client.get("/config")
        response.raise_for_status()
        return response.json()

    async def update_config(self, config: Dict[str, Any]) -> Dict[str, Any]:
        """更新配置"""
        response = await self._client.patch("/config", json=config)
        response.raise_for_status()
        return response.json()

    async def list_providers(self) -> Dict[str, Any]:
        """列出可用模型提供商"""
        response = await self._client.get("/config/providers")
        response.raise_for_status()
        return response.json()

    # ==================== MCP 管理 ====================

    async def get_mcp_status(self) -> Dict[str, Any]:
        """获取 MCP 服务器状态"""
        response = await self._client.get("/mcp")
        response.raise_for_status()
        return response.json()

    async def add_mcp_server(self, name: str, config: Dict[str, Any]) -> Dict[str, Any]:
        """添加 MCP 服务器"""
        response = await self._client.post(
            "/mcp",
            json={"name": name, "config": config}
        )
        response.raise_for_status()
        return response.json()

    async def connect_mcp_server(self, name: str) -> bool:
        """连接 MCP 服务器"""
        response = await self._client.post(f"/mcp/{name}/connect")
        response.raise_for_status()
        return response.json()

    async def disconnect_mcp_server(self, name: str) -> bool:
        """断开 MCP 服务器"""
        response = await self._client.post(f"/mcp/{name}/disconnect")
        response.raise_for_status()
        return response.json()

    # ==================== 系统 API ====================

    async def health_check(self) -> Dict[str, Any]:
        """健康检查"""
        response = await self._client.get("/global/health")
        response.raise_for_status()
        return response.json()

    async def get_paths(self) -> Dict[str, str]:
        """获取路径信息"""
        response = await self._client.get("/path")
        response.raise_for_status()
        return response.json()

    async def list_agents(self) -> List[Dict[str, Any]]:
        """列出可用 Agent"""
        response = await self._client.get("/agent")
        response.raise_for_status()
        return response.json()

    async def list_skills(self) -> List[Dict[str, Any]]:
        """列出可用技能"""
        response = await self._client.get("/skill")
        response.raise_for_status()
        return response.json()

    async def dispose_instance(self) -> bool:
        """释放实例"""
        response = await self._client.post("/instance/dispose")
        response.raise_for_status()
        return response.json()

    # ==================== 资源清理 ====================

    async def close(self) -> None:
        """关闭客户端"""
        await self._client.aclose()
```
