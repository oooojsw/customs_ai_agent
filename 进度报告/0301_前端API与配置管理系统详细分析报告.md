    // 5. 处理Azure界面显隐

}

const res = await fetch(CHAT_URL, {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify({ message: msg, session_id: SESSION_ID, language: window.currentLanguage })
});

const reader = res.body.getReader();
const decoder = new TextDecoder();

while(true) {
    const {done, value} = await reader.read();
    if(done) break;
    
    buffer += decoder.decode(value, {stream: true});
    const lines = buffer.split('\n\n');
    
    for (const line of lines) {
        if (line.startsWith('data: ')) {
            const data = JSON.parse(line.substring(6));
            
            // 处理不同类型的消息
            if (data.type === 'answer') { /* 文本回答 */ }
            if (data.type === 'thinking') { /* 思考过程 */ }
            if (data.type === 'tool_start') { /* 工具调用开始 */ }
            if (data.type === 'tool_end') { /* 工具调用结束 */ }
        }
    }
}

function handleIndexEvent(data) {
    switch (data.type) {
        case 'init':
            // 开始扫描文件
            break;
        case 'progress':
            // 文件处理进度
            updateCurrentFile(`正在处理: ${data.current_file}`);
            break;
        case 'embedding_start':
            // 开始向量化
            break;
        case 'embedding_progress':
            // 向量化进度
            updateEmbeddingProgress(data.batch_num, data.total_batches, data.percentage);
            break;
        case 'complete':
            showCompleteUI(data.stats);
            break;
        case 'error':
            showErrorUI(data.message);
            break;
    }
}

## 八、安全机制

### 8.1 前端Key保护

1. **输入框脱敏**: 使用 `type="password"`
2. **Base URL验证**: 加载时检查域名是否匹配
3. **竞态条件防御**: 见2.6节
4. **不保存脱敏值**: 避免加载 `sk-......`

### 8.2 请求安全

1. **Content-Type声明**: `headers: { 'Content-Type': 'application/json' }`
2. **URL编码**: `encodeURIComponent(apiKey)` 防止特殊字符问题
3. **缓存防抖**: 添加时间戳 `?_t=${Date.now()}`
