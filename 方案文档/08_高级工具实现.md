### 12.1 文件智能处理器

```python
# src/services/opencode/tools/file_processor.py

import json
import csv
from io import StringIO
from typing import Dict, Any


class FileProcessorTool:
    """文件智能处理器 - 读取/分析多种格式文件"""

    name = "opencode_file_processor"
    description = """[OpenCode文件处理] 智能文件处理器。

支持格式: CSV, JSON, XML, TXT, Markdown, YAML
操作: analyze(分析), read(读取), extract(提取), info(信息)

参数 (JSON):
{
  "action": "analyze|read|extract|info",
  "file_path": "data/report.csv",
  "options": {
    "output_format": "json|text",
    "encoding": "utf-8",
    "extract_columns": ["col1", "col2"]
  }
}"""

    SUPPORTED_FORMATS = {
        ".csv": "csv", ".json": "json", ".xml": "xml",
        ".txt": "text", ".md": "markdown", ".yaml": "yaml", ".yml": "yaml"
    }

    async def execute(self, params: Dict[str, Any], client) -> Dict[str, Any]:
        action = params.get("action", "read")
        file_path = params.get("file_path")

        if not file_path:
            return {"success": False, "error": "缺少 file_path 参数"}

        try:
            content = await client.read_file(file_path)
            content_text = content.get("content", "") if isinstance(content, dict) else str(content)

            if action == "read":
                return {"success": True, "content": content_text, "size": len(content_text)}

            elif action == "analyze":
                ext = "." + file_path.rsplit(".", 1)[-1] if "." in file_path else ""
                analysis = {
                    "success": True,
                    "file_path": file_path,
                    "file_type": self.SUPPORTED_FORMATS.get(ext, "unknown"),
                    "size": len(content_text),
                    "lines": content_text.count('\n') + 1,
                    "preview": content_text[:500]
                }
                if ext == ".csv":
                    analysis["csv_analysis"] = self._analyze_csv(content_text)
                elif ext == ".json":
                    analysis["json_analysis"] = self._analyze_json(content_text)
                return analysis

            elif action == "info":
                return {
                    "success": True,
                    "file_path": file_path,
                    "extension": ext,
                    "type": self.SUPPORTED_FORMATS.get(ext, "unknown")
                }

            else:
                return {"success": False, "error": f"不支持的操作: {action}"}

        except Exception as e:
            return {"success": False, "error": str(e)}

    def _analyze_csv(self, content: str) -> Dict[str, Any]:
        try:
            reader = csv.DictReader(StringIO(content))
            rows = list(reader)
            return {
                "columns": reader.fieldnames or [],
                "row_count": len(rows),
                "sample": rows[:3] if rows else []
            }
        except Exception as e:
            return {"error": str(e)}

    def _analyze_json(self, content: str) -> Dict[str, Any]:
        try:
            data = json.loads(content)
            if isinstance(data, dict):
                return {"type": "object", "keys": list(data.keys())[:20], "depth": 1}
            elif isinstance(data, list):
                return {"type": "array", "length": len(data)}
            return {"type": type(data).__name__}
        except json.JSONDecodeError as e:
            return {"error": str(e)}
```

### 12.2 代码执行工具

```python
# src/services/opencode/tools/code_executor.py

import json
from typing import Dict, Any


class CodeExecutorTool:
    """代码执行工具 - 通过 OpenCode 执行 Python/Shell"""

    name = "opencode_code_executor"
    description = """[OpenCode代码执行] 执行 Python 脚本或 Shell 命令。

参数 (JSON):
{
  "language": "python|bash|shell",
  "code": "print('hello')",
  "timeout": 30,
  "args": {"key": "value"}
}

安全限制:
- 禁止 rm -rf, format, del 等危险命令
- 禁止访问系统敏感目录
- 执行超时自动终止
"""

    async def execute(self, params: Dict[str, Any], client) -> Dict[str, Any]:
        language = params.get("language", "python")
        code = params.get("code", "")
        timeout = params.get("timeout", 30)

        if not code:
            return {"success": False, "error": "缺少 code 参数"}

        # 危险命令检查
        dangerous_patterns = ["rm -rf", "format", "del /s", "sudo", "chmod 777"]
        code_lower = code.lower()
        for pattern in dangerous_patterns:
            if pattern in code_lower:
                return {"success": False, "error": f"检测到危险命令模式: {pattern}"}

        try:
            # 创建临时会话执行
            session = await client.create_session(workspace_dir=".")

            # 根据语言构建命令
            if language == "python":
                command = f"python -c \"{code.replace('\"', '\\\"')}\""
            else:
                command = code

            result = await client.send_message(
                session_id=session["id"],
                parts=[{"type": "text", "text": f"执行命令: {command}"}]
            )

            # 清理会话
            await client.delete_session(session["id"])

            return {
                "success": True,
                "language": language,
                "output": str(result)[:5000]
            }

        except Exception as e:
            return {"success": False, "error": str(e)}
```

### 12.3 批量处理工具

```python
# src/services/opencode/tools/batch_processor.py

import json
import asyncio
from typing import Dict, Any, List


class BatchProcessorTool:
    """批量处理工具 - 并行执行多个操作"""

    name = "opencode_batch_processor"
    description = """[OpenCode批量处理] 并行执行多个文件操作。

参数 (JSON):
{
  "operations": [
    {"action": "read", "file_path": "data/file1.csv"},
    {"action": "read", "file_path": "data/file2.json"},
    {"action": "search", "pattern": "keyword"}
  ],
  "parallel": true,
  "max_concurrent": 5
}
"""

    async def execute(self, params: Dict[str, Any], client) -> Dict[str, Any]:
        operations = params.get("operations", [])
        parallel = params.get("parallel", True)
        max_concurrent = params.get("max_concurrent", 5)

        if not operations:
            return {"success": False, "error": "缺少 operations 参数"}

        try:
            if parallel:
                semaphore = asyncio.Semaphore(max_concurrent)

                async def run_op(op):
                    async with semaphore:
                        return await self._execute_single(op, client)

                results = await asyncio.gather(
                    *[run_op(op) for op in operations],
                    return_exceptions=True
                )
            else:
                results = []
                for op in operations:
                    results.append(await self._execute_single(op, client))

            return {
                "success": True,
                "total": len(operations),
                "results": [
                    r if isinstance(r, dict) else {"error": str(r)}
                    for r in results
                ]
            }

        except Exception as e:
            return {"success": False, "error": str(e)}

    async def _execute_single(self, op: Dict[str, Any], client) -> Dict[str, Any]:
        action = op.get("action", "")

        if action == "read":
            return await client.read_file(op["file_path"])
        elif action == "search":
            return await client.search_text(op["pattern"])
        elif action == "list":
            return await client.list_files(op.get("path", "."))
        else:
            return {"error": f"不支持的操作: {action}"}
```
