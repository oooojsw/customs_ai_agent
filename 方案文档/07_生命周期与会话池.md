### 9.2 服务管理器

```python
# src/services/opencode/manager.py

import asyncio
import os
import time
from typing import Optional, Dict, Any, Callable
from pathlib import Path
from enum import Enum
from dataclasses import dataclass, field

import httpx


class ServiceState(Enum):
    """服务状态"""
    UNINITIALIZED = "uninitialized"
    STARTING = "starting"
    READY = "ready"
    STOPPING = "stopping"
    STOPPED = "stopped"
    ERROR = "error"


@dataclass
class OpenCodeServiceConfig:
    """OpenCode 服务配置"""
    port: int = 4096
    host: str = "127.0.0.1"
    opencode_source_dir: str = ""  # OpenCode 源码目录（用 bun run dev serve 启动）
    workspace_dir: str = ""
    startup_timeout: int = 60
    health_check_interval: int = 30
    max_idle_time: int = 3600
    max_retries: int = 3
    max_concurrent_sessions: int = 5
    session_pool_size: int = 3
    security_enabled: bool = True
    allowed_dirs: list = field(
        default_factory=lambda: ["data/", "config/", "temp/"]
    )


class OpenCodeService:
    """OpenCode 服务管理器（单例）"""

    _instance: Optional['OpenCodeService'] = None
    _lock = asyncio.Lock()

    def __new__(cls, *args, **kwargs):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    def __init__(self, config: Optional[OpenCodeServiceConfig] = None):
        if hasattr(self, '_initialized'):
            return

        self.config = config or OpenCodeServiceConfig()
        self._state = ServiceState.UNINITIALIZED
        self._process = None
        self._client = None
        self._start_time = None
        self._health_check_task = None
        self._state_change_callbacks = []
        self._initialized = True

    @classmethod
    def get_instance(cls) -> 'OpenCodeService':
        return cls._instance or cls()

    # ==================== 生命周期 ====================

    async def start(self) -> bool:
        """启动服务"""
        async with self._lock:
            if self._state in [ServiceState.STARTING, ServiceState.READY]:
                return True

            self._set_state(ServiceState.STARTING)

            try:
                await self._prepare_workspace()
                opencode_dir = self._get_source_dir()
                if not opencode_dir.exists():
                    raise FileNotFoundError(f"OpenCode 源码目录不存在: {opencode_dir}")

                await self._start_process(opencode_dir)
                await self._wait_for_ready()

                self._client = httpx.AsyncClient(
                    base_url=self.base_url,
                    timeout=httpx.Timeout(30.0),
                    proxies={"http://": None, "https://": None}
                )

                await self._start_health_check()
                self._start_time = time.time()
                self._set_state(ServiceState.READY)

                print(f"[OpenCodeService] ✅ 服务启动成功: {self.base_url}")
                return True

            except Exception as e:
                print(f"[OpenCodeService] ❌ 启动失败: {e}")
                await self.stop()
                self._set_state(ServiceState.ERROR)
                return False

    async def stop(self):
        """停止服务"""
        async with self._lock:
            self._set_state(ServiceState.STOPPING)

            if self._health_check_task:
                self._health_check_task.cancel()

            if self._client:
                await self._client.aclose()
                self._client = None

            if self._process:
                try:
                    self._process.terminate()
                    await asyncio.wait_for(self._process.wait(), timeout=5.0)
                except asyncio.TimeoutError:
                    self._process.kill()
                finally:
                    self._process = None

            self._start_time = None
            self._set_state(ServiceState.STOPPED)

    async def restart(self):
        """重启服务"""
        await self.stop()
        await asyncio.sleep(2)
        await self.start()

    # ==================== 内部方法 ====================

    def _set_state(self, new_state: ServiceState):
        old_state = self._state
        self._state = new_state
        for cb in self._state_change_callbacks:
            try:
                cb(old_state, new_state)
            except Exception:
                pass

    def _get_source_dir(self) -> Path:
        """获取 OpenCode 源码目录"""
        return Path(self.config.opencode_source_dir)

    async def _prepare_workspace(self):
        """准备工作空间"""
        workspace = Path(self.config.workspace_dir)
        workspace.mkdir(parents=True, exist_ok=True)

    async def _start_process(self, opencode_dir: Path):
        """启动 OpenCode 服务器（通过 bun run dev serve）"""
        cmd = ["bun", "run", "dev", "serve", "--port", str(self.config.port)]
        env = os.environ.copy()
        for k in ["HTTP_PROXY", "HTTPS_PROXY", "http_proxy", "https_proxy"]:
            env.pop(k, None)

        self._process = await asyncio.create_subprocess_exec(
            *cmd,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
            cwd=str(opencode_dir),  # 关键：cwd 设为 OpenCode 源码目录
            env=env
        )

    async def _wait_for_ready(self, timeout: int = 60):
        start = time.time()
        while time.time() - start < timeout:
            try:
                async with httpx.AsyncClient(timeout=5.0) as c:
                    r = await c.get(f"{self.base_url}/global/health")
                    if r.status_code == 200:
                        return
            except Exception:
                pass
            await asyncio.sleep(1)
        raise TimeoutError(f"服务未在 {timeout} 秒内就绪")

    async def _start_health_check(self):
        async def loop():
            while True:
                try:
                    await asyncio.sleep(self.config.health_check_interval)
                    if self._state != ServiceState.READY:
                        continue
                    if not await self._check_health():
                        print("[OpenCodeService] ⚠️ 健康检查失败，尝试重启...")
                        await self.restart()
                except asyncio.CancelledError:
                    break
        self._health_check_task = asyncio.create_task(loop())

    async def _check_health(self) -> bool:
        try:
            if not self._client:
                return False
            r = await self._client.get("/global/health")
            return r.status_code == 200
        except Exception:
            return False

    @property
    def state(self) -> ServiceState:
        return self._state

    @property
    def is_ready(self) -> bool:
        return self._state == ServiceState.READY

    @property
    def base_url(self) -> str:
        return f"http://{self.config.host}:{self.config.port}"

    def get_client(self) -> httpx.AsyncClient:
        if not self._client:
            raise RuntimeError("客户端未初始化")
        return self._client
```

### 10.1 会话池设计

```python
# src/services/opencode/session_pool.py

import time
from typing import Dict, Any, Optional


class SessionPool:
    """会话池管理器"""

    def __init__(self, max_sessions: int = 10):
        self.max_sessions = max_sessions
        self.sessions: Dict[str, Dict[str, Any]] = {}

    async def get_or_create(self, client, workspace_dir: str) -> str:
        """获取空闲会话或创建新会话"""
        # 查找空闲会话
        for sid, info in self.sessions.items():
            if info["directory"] == workspace_dir and info["active"]:
                return sid

        # 超出限制，清理最旧的
        if len(self.sessions) >= self.max_sessions:
            oldest = min(self.sessions.items(), key=lambda x: x[1]["created"])
            await client.delete_session(oldest[0])
            del self.sessions[oldest[0]]

        # 创建新会话
        result = await client.create_session(workspace_dir=workspace_dir)
        session_id = result["id"]
        self.sessions[session_id] = {
            "directory": workspace_dir,
            "created": time.time(),
            "active": True
        }
        return session_id

    async def release(self, session_id: str):
        """释放会话"""
        if session_id in self.sessions:
            self.sessions[session_id]["active"] = True

    async def cleanup(self, client):
        """清理所有会话"""
        for sid in list(self.sessions.keys()):
            try:
                await client.delete_session(sid)
            except Exception:
                pass
        self.sessions.clear()
```

### 11.1 事件订阅实现

```python
# src/services/opencode/events.py

import json
from typing import AsyncIterator, Dict, Any, Optional


class OpenCodeEventListener:
    """OpenCode SSE 事件监听器"""

    def __init__(self, client):
        self.client = client

    async def subscribe(
        self,
        directory: Optional[str] = None
    ) -> AsyncIterator[Dict[str, Any]]:
        """
        订阅 SSE 事件流

        支持的事件类型：
        - session.created
        - session.updated
        - session.deleted
        - message.updated
        - message.part.updated
        - file.edited
        - permission.asked
        - question.asked
        """
        params = {}
        if directory:
            params["directory"] = directory

        async with self.client._client.stream(
            "GET", "/event", params=params
        ) as response:
            response.raise_for_status()
            event_type = None

            async for line in response.aiter_lines():
                if not line.strip():
                    continue

                if line.startswith("event:"):
                    event_type = line[6:].strip()
                    continue

                if line.startswith("data:"):
                    data_str = line[5:].strip()
                    try:
                        event_data = json.loads(data_str)
                        yield {
                            "type": event_type,
                            "properties": event_data
                        }
                    except json.JSONDecodeError:
                        continue
```
