#### 6.2.2 修改 src/main.py

在 `lifespan` 函数中添加：

```python
# ==================== OpenCode 集成 ====================
from src.config.opencode_config import opencode_config_loader
from src.services.opencode_client import OpenCodeClient
from src.services.opencode_tools import OpenCodeTools

# 加载 OpenCode 配置
opencode_config = opencode_config_loader.get_config()

if opencode_config.enabled:
    print("🔄 [OpenCode] 正在启动 OpenCode 服务器...")

    # 1. 启动 OpenCode 服务器（通过 bun run dev serve）
    import subprocess
    import asyncio

    opencode_process = None
    try:
        # 检查 OpenCode 源码目录
        opencode_dir = Path(opencode_config.opencode_source_dir)
        if not opencode_dir.exists():
            raise FileNotFoundError(f"OpenCode 源码目录不存在: {opencode_dir}")

        # 构建启动命令：bun run dev serve --port 4096 --hostname 127.0.0.1
        cmd = [
            "bun", "run", "dev", "serve",
            "--port", str(opencode_config.port),
            "--hostname", opencode_config.hostname
        ]

        # 设置环境变量
        env = os.environ.copy()
        if opencode_config.server_password:
            env["OPENCODE_SERVER_PASSWORD"] = opencode_config.server_password
        if opencode_config.server_username:
            env["OPENCODE_SERVER_USERNAME"] = opencode_config.server_username
        # 清除代理，避免干扰本地连接
        for key in ["HTTP_PROXY", "HTTPS_PROXY", "http_proxy", "https_proxy"]:
            env.pop(key, None)

        # 启动进程（cwd 设为 OpenCode 源码目录）
        opencode_process = subprocess.Popen(
            cmd,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            env=env,
            cwd=str(opencode_dir)
        )

        # 等待服务器启动
        await asyncio.sleep(5)

        print(f"✅ [OpenCode] 服务器已启动 (PID: {opencode_process.pid})")

    except Exception as e:
        print(f"❌ [OpenCode] 服务器启动失败: {e}")
        opencode_process = None

    # 2. 创建客户端
    try:
        base_url = f"http://{opencode_config.hostname}:{opencode_config.port}"
        opencode_client = OpenCodeClient(
            base_url=base_url,
            username=opencode_config.server_username,
            password=opencode_config.server_password,
            timeout=opencode_config.timeout,
            proxy=opencode_config.http_proxy
        )

        # 健康检查
        health = await opencode_client.health_check()
        print(f"✅ [OpenCode] 客户端连接成功: {health}")

        # 保存到 app.state
        app.state.opencode_client = opencode_client
        app.state.opencode_process = opencode_process

    except Exception as e:
        print(f"❌ [OpenCode] 客户端连接失败: {e}")
        app.state.opencode_client = None
        app.state.opencode_process = None
else:
    print("ℹ️ [OpenCode] 集成已禁用")
    app.state.opencode_client = None
    app.state.opencode_process = None
```

#### 6.3.2 工作目录限制

OpenCode 的工作目录由 `x-opencode-directory` 请求头或 `directory` 查询参数指定，建议在客户端层面限制为项目目录：

```python
# 在 opencode_client.py 中添加验证
def _validate_workspace_dir(self, path: str) -> str:
    """验证并规范化工作目录"""
    from pathlib import Path
    resolved = Path(path).resolve()
    project_root = Path(__file__).resolve().parent.parent.parent

    if not str(resolved).startswith(str(project_root)):
        raise ValueError(f"工作目录必须在项目内: {path}")

    return str(resolved)
```

### 6.4 错误处理

#### 6.4.1 连接错误

```python
async def _ensure_connection(self):
    """确保连接可用"""
    try:
        health = await self.client.health_check()
        if not health.get("healthy"):
            raise ConnectionError("OpenCode 服务器不健康")
    except httpx.ConnectError:
        raise ConnectionError("无法连接到 OpenCode 服务器")
    except httpx.TimeoutException:
        raise TimeoutError("OpenCode 服务器响应超时")
```

#### 6.4.2 会话管理

```python
# 会话池管理
class SessionPool:
    def __init__(self, max_sessions: int = 10):
        self.max_sessions = max_sessions
        self.sessions: Dict[str, Dict[str, Any]] = {}

    async def get_or_create(self, workspace_dir: str) -> str:
        """获取或创建会话"""
        # 查找可用会话
        for sid, info in self.sessions.items():
            if info["directory"] == workspace_dir and info["active"]:
                return sid

        # 创建新会话
        if len(self.sessions) >= self.max_sessions:
            # 清理最旧的会话
            oldest = min(self.sessions.items(), key=lambda x: x[1]["created"])
            await self.client.delete_session(oldest[0])
            del self.sessions[oldest[0]]

        result = await self.client.create_session(workspace_dir)
        session_id = result["id"]
        self.sessions[session_id] = {
            "directory": workspace_dir,
            "created": time.time(),
            "active": True
        }
        return session_id
```

### 6.6 故障排除

| 问题           | 原因           | 解决方案                           |
| -------------- | -------------- | ---------------------------------- |
| 服务器启动失败 | 端口被占用     | 更换端口或关闭占用进程             |
| 连接超时       | 服务器未启动   | 检查启动日志，确认服务器状态       |
| 认证失败       | 密码不匹配     | 检查 OPENCODE_SERVER_PASSWORD 配置 |
| 文件操作失败   | 工作目录不正确 | 检查 workspace_dir 配置            |
| 工具调用失败   | 客户端未初始化 | 检查 main.py 中的初始化逻辑        |
